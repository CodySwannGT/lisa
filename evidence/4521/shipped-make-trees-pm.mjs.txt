function npmLock(extra) {
  const packages = {
    "": { name: "verify-host" },
    "node_modules/zod": { version: "4.6.6", resolved: "https://registry.npmjs.org/zod/-/zod-4.6.6.tgz" },
    "node_modules/lint-plugin": { version: "1.0.0", dev: true, dependencies: { zod: "^3.22.4" } },
    "node_modules/lint-plugin/node_modules/zod": { version: "3.25.76", dev: true, resolved: "https://registry.npmjs.org/zod/-/zod-3.25.76.tgz" },
  };
  if (extra === "uncovered") {
    packages["node_modules/app-dep"] = { version: "2.0.0", dependencies: { zod: "3.25.76" } };
    packages["node_modules/app-dep/node_modules/zod"] = { version: "3.25.76", resolved: "https://registry.npmjs.org/zod/-/zod-3.25.76.tgz" };
  }
  if (extra === "alias")
    packages["node_modules/schema-alias"] = { name: "zod", version: "3.25.76", resolved: "https://registry.npmjs.org/zod/-/zod-3.25.76.tgz" };
  return JSON.stringify({ name: "verify-host", lockfileVersion: 3, requires: true, packages }, null, 2) + "\n";
}

function yarnLock(extra) {
  const blocks = [
    'lint-plugin@1.0.0:\n  version "1.0.0"\n  resolved "https://registry.yarnpkg.com/lint-plugin/-/lint-plugin-1.0.0.tgz"\n  dependencies:\n    zod "^3.22.4"',
    'zod@^3.22.4:\n  version "3.25.76"\n  resolved "https://registry.yarnpkg.com/zod/-/zod-3.25.76.tgz"',
    'zod@^4.6.6:\n  version "4.6.6"\n  resolved "https://registry.yarnpkg.com/zod/-/zod-4.6.6.tgz"',
  ];
  if (extra === "uncovered") {
    blocks.push('app-dep@2.0.0:\n  version "2.0.0"\n  resolved "https://registry.yarnpkg.com/app-dep/-/app-dep-2.0.0.tgz"\n  dependencies:\n    zod "3.25.76"');
    blocks.push('zod@3.25.76:\n  version "3.25.76"\n  resolved "https://registry.yarnpkg.com/zod/-/zod-3.25.76.tgz"');
  }
  if (extra === "alias")
    blocks.push('"schema-alias@npm:zod@3.25.76":\n  version "3.25.76"\n  resolved "https://registry.yarnpkg.com/zod/-/zod-3.25.76.tgz"');
  return `# THIS IS AN AUTOGENERATED FILE. DO NOT EDIT THIS FILE DIRECTLY.\n# yarn lockfile v1\n\n\n${blocks.join("\n\n")}\n`;
}

const listed = { npm: "node_modules/lint-plugin/node_modules/zod", yarn: "zod@^3.22.4" };
const cases = [
  ["control-no-exclusion", "base", false, "fail"],
  ["covered-only", "base", true, "pass"],
  ["uncovered-production", "uncovered", true, "fail"],
  ["npm-alias", "alias", true, "fail"],
];

rmSync(outRoot, { recursive: true, force: true });
mkdirSync(outRoot, { recursive: true });
writeFileSync(join(outRoot, "npm-audit-report.json"), JSON.stringify(npmReport));
writeFileSync(join(outRoot, "yarn-audit-report.ndjson"), yarnReport);
const manifest = [];
for (const manager of ["npm", "yarn"]) {
  for (const [name, layout, scoped, expect] of cases) {
    const id = `${manager}-${name}`;
    const root = join(outRoot, id);
    mkdirSync(join(root, "scripts"), { recursive: true });
    mkdirSync(join(root, "node_modules", "@codyswann"), { recursive: true });
    writeFileSync(
      join(root, "package.json"),
      JSON.stringify({
        name: "verify-host",
        private: true,
        type: "module",
        scripts: {
          "security:audit": "$npm_execpath run security:audit:lisa",
          "security:audit:lisa": "node scripts/lisa-audit.mjs",
        },
      }, null, 2)
    );
    copyFileSync(join(pkgRoot, "all/copy-overwrite/scripts/lisa-audit.mjs"), join(root, "scripts/lisa-audit.mjs"));
    symlinkSync(pkgRoot, join(root, "node_modules/@codyswann/lisa"));
    if (manager === "npm") writeFileSync(join(root, "package-lock.json"), npmLock(layout));
    else writeFileSync(join(root, "yarn.lock"), yarnLock(layout));
    const exclusions = scoped
      ? [{ id: GHSA, package: "zod", versions: ">=3.0.0 <4.0.0", paths: [listed[manager]], reason: REASON }]
      : [];
    writeFileSync(join(root, "audit.ignore.local.json"), JSON.stringify({ exclusions }, null, 2));
    const off = { push: "off" };
    writeFileSync(join(root, ".lisa.config.json"), JSON.stringify({
      gates: {
        runner: "bun run",
        "dependency-vulnerability": { push: "required" },
        "coverage-adequacy": off, "test-correctness": off, "test-integration": off,
        traceability: off, "type-correctness": off,
      },
    }, null, 2));
    manifest.push(`${id}\t${manager}\t${expect}`);
  }
}
writeFileSync(join(outRoot, "cases.tsv"), `${manifest.join("\n")}\n`);
